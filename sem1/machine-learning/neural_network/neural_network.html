def plot_decision_boundary(X, y, model, title=""):
    h = 0.05
    x_min, x_max = X[:,0].min() - 1, X[:,0].max() + 1
    y_min, y_max = X[:,1].min() - 1, X[:,1].max() + 1
    xx, yy = np.meshgrid(
        np.arange(x_min, x_max, h),
        np.arange(y_min, y_max, h)
    )
    pts = np.c_[xx.ravel(), yy.ravel()]

    Z = model.predict(pts)

    Z = Z.reshape(xx.shape)

    # decition boundary
    plt.contourf(xx, yy, Z, alpha=0.3)

    # decision line
    plt.contour(xx, yy, Z, colors='k', linewidths=1)

    scatter = plt.scatter(X[:,0], X[:,1], c=y, edgecolors='k')

    plt.legend(*scatter.legend_elements(), loc="upper right", title="Classes")
    plt.title(title)
    plt.xlabel('Feature 1'); plt.ylabel('Feature 2')
    plt.show()

class NeuralNetwork(BaseEstimator, ClassifierMixin):
    def __init__(self, layers=[2, 10, 2], lr = 0.01, epochs = 100, epsilon=1e-5, batch_size = None, hidden_activation='relu'):
        self.layers = layers
        self.lr = lr
        self.epochs = epochs
        self.epsilon = epsilon
        self.batch_size = batch_size
        self.hidden_activation = hidden_activation
        self.W = []
        self.b = []
        for i in range(len(layers) - 1):
            self.W.append(np.random.randn(layers[i], layers[i+1]) * np.sqrt(2/layers[i]))
            self.b.append(np.zeros((1, layers[i+1]))) 
            
    def relu(self, x):
        return np.maximum(0, x)
    
    def relu_derivative(self, x):
        return (x > 0 ).astype(float)
    
    def sigmoid(self, x):
        return 1 / (1 + np.exp(-x))
    
    def sigmoid_derivative(self, x):
        sig = self.sigmoid(x)
        return sig * (1 - sig)
    
    def softmax(self, x):
        exp_x = np.exp(x - np.max(x, axis=1, keepdims=True))
        return exp_x / np.sum(exp_x, axis=1, keepdims=True)
    
    def forward(self, X):
        self.cache = []
        a = X # input layer
        for i in range(len(self.W)):
            s = a @ self.W[i] + self.b[i] # 1st iteration: (batch_size, num_features) @ (2, 10) + (1, 10) = (batch_size, 10)
            self.cache.append({'a': a, 's': s})
            if i < len(self.W) - 1:
                if self.hidden_activation == 'relu':
                    a = self.relu(s)
                elif self.hidden_activation == 'sigmoid':
                    a = self.sigmoid(s)
                else:
                    raise ValueError("Unsupported hidden activation function. Use 'relu' or 'sigmoid'.")
            else:
                a = self.softmax(s)
        return a
    
    def backward(self, y_true, output):
        grads_W = [None]*len(self.W)
        grads_b = [None]*len(self.W)
        eta = output - y_true
        
        for i in reversed(range(len(self.W))):
            a_prev = self.cache[i]['a'] # output of previous layer -> input to this layer
            s = self.cache[i]['s'] # output (result) of this layer before activation
            
            if i == len(self.W) - 1:
                ds = eta
            else:
                if self.hidden_activation == 'relu':
                    ds = eta * self.relu_derivative(s)
                elif self.hidden_activation == 'sigmoid':
                    ds = eta * self.sigmoid_derivative(s)
                else:
                    raise ValueError("Unsupported hidden activation function. Use 'relu' or 'sigmoid'.")
            
            # (num_features, num_classes) = (num_features, batch_size) @ (batch_size, num_classes) - for the last layer
            # (n_in, n_out) = (n_in, batch_size) @ (batch_size, n_out) - for the hidden layers
            grads_W[i] = a_prev.T @ ds / a_prev.shape[0]
            grads_b[i] = np.sum(ds, axis=0, keepdims=True) / a_prev.shape[0]
            
            if i > 0:
                # (batch_size, n_in) = (batch_size, n_out) @ (n_out, n_in)
                eta = ds @ self.W[i].T 
                
        return grads_W, grads_b
    
    def adjust(self, grads_W, grads_b):
        for i in range(len(self.W)):
            self.W[i] -= self.lr * grads_W[i]
            self.b[i] -= self.lr * grads_b[i]
    
    def fit(self, X, y):
        num_classes = len(np.unique(y))
        y_onehot = np.eye(num_classes)[y]
        prev_loss = None
        n_samples = X.shape[0]
        batch_size = self.batch_size if self.batch_size is not None else n_samples 
        
        for epoch in range(self.epochs):
            indices = np.arange(n_samples)
            np.random.shuffle(indices)
            X_shuffled = X[indices]
            y_onehot_shuffled = y_onehot[indices]
            
            for start in range(0, n_samples, batch_size):
                end = start + batch_size
                X_batch = X_shuffled[start:end]
                y_batch = y_onehot_shuffled[start:end]

                out = self.forward(X_batch) # (batch_size,num_classes)
                grads_W, grads_b = self.backward(y_batch, out)
                self.adjust(grads_W, grads_b)
            
            out_ = self.forward(X)
            # loss = np.mean((y_onehot - out_)**2) # MSE 
            loss = -np.mean(np.sum(y_onehot * np.log(out_ + 1e-8), axis=1)) # Cross-entropy
            if epoch % 50 == 0:
                # print(f"Epoch {epoch} - Loss: {loss}")
                pass
            if prev_loss is not None and abs(prev_loss - loss) < self.epsilon:
                print(f"Loss diff < epsilon at epoch {epoch} - Loss {loss}")
                break
            prev_loss = loss
    
    def predict(self, X):
        out = self.forward(X)
        return np.argmax(out, axis=1) # index of the largest value in each row
    
    def predict_proba(self, X):
        out = self.forward(X)
        return out
